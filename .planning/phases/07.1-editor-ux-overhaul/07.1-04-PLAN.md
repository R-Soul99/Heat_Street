```ts
export interface ShellTool { id: string; label: string; icon: string; keyLabel?: string; tip: string; disabled?: boolean }
export type MenuItemSpec =
  | { id: string; label: string; tip: string; shortcut?: string; run: () => void;
      checked?: () => boolean; enabled?: () => boolean }
  | { separator: true };
export interface MenuSpec { label: string; tip: string; items: MenuItemSpec[] }
export interface InfoField { key: string; label: string; tip: string }
export interface ShellOptions {
  root: HTMLElement;                 // #app; the editor's canvas is already inside it
  menus: MenuSpec[];                 // File/Area, Import, View, Help (Help gets the tooltip toggle appended automatically)
  tools: ShellTool[];
  onToolChange: (id: string) => void;
  history: { undo: () => void; redo: () => void; clear: () => void; clearLabel: string; clearTip: string };
  view: { zoomIn: () => void; zoomOut: () => void; fit: () => void; pan: (dx: -1 | 0 | 1, dy: -1 | 0 | 1) => void;
          extra?: { id: string; label: string; tip: string; run: () => void; pressed?: () => boolean }[] };
  infoFields: InfoField[];
}
export interface Shell {
  setActiveTool(id: string): void;    // highlights rail button, calls onToolChange, clears + shows flyout
  flyout: HTMLElement;                // editor fills it in onToolChange; empty => flyout hidden
  info: { set(key: string, text: string): void; status(text: string, kind?: "info" | "ok" | "error"): void; setDirty(dirty: boolean): void };
  setHistoryState(canUndo: boolean, canRedo: boolean): void;
  setToolEnabled(id: string, enabled: boolean): void;
  refreshMenus(): void;               // re-evaluates checked/enabled
  isDialogOpen(): boolean;
}
```
