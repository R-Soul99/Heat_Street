export interface RoutineResult {
  id: string; label: string;
  value: number; unit: string;
  target: string;      // human-readable, e.g. "6.0-7.0 s"
  pass: boolean;
}
export interface Routine {
  id: string; label: string;
  /** Input for this tick given the current state; return null to end the routine. */
  drive(tick: number, s: VehicleSample): InputFrame | null;
  /** Accumulate measurements each tick. */
  sample(tick: number, s: VehicleSample, acc: Record<string, number>): void;
  evaluate(acc: Record<string, number>): RoutineResult;
  /** Optional one-off setup, e.g. the ramp routine's scripted launch. */
  setup?(vehicle: Vehicle): void;
}
export const ROUTINES: readonly Routine[]        // accel, brake, skidpad, slalom, ramp, stability
export function runRoutine(routine: Routine, tuning: VehicleTuning): RoutineResult
export function runAllRoutines(tuning: VehicleTuning): RoutineResult[]
</interfaces>

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| live-tuned `VehicleTuning` -> throwaway world | The browser panel passes a user-mutated tuning object into `runRoutine` |
| telemetry world -> WASM linear memory | Every run allocates a Rapier world that JS GC cannot reclaim |
| telemetry world -> live game world | A telemetry run must never perturb the play session |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-02-06 | Denial of Service | `runRoutine` world lifetime | mitigate | `world.free()` inside a `finally` so a throwing routine cannot leak; `world.free()` releases the world's bodies and colliders too. Without it a ~30-routine tuning session holds 30 live worlds against the 4 GB WASM ceiling (Pitfall 13). |
| T-02-17 | Denial of Service | a routine that never returns `null` | mitigate | Explicit `MAX_TICKS` cap; the loop terminates and `evaluate` reports the incomplete run as a failure rather than hanging the browser frame. |
| T-02-18 | Tampering | telemetry perturbing the live simulation | mitigate | `runRoutine` always builds its own `createWorld()` and never receives the game world; asserted by the fact the Vitest suite constructs no game scene at all. A telemetry run can never pollute a future medal time. |
| T-02-01 | Tampering / DoS | hostile tuning values reaching the telemetry solver | mitigate | The panel clamps through `clampTuning` (plan 02-01) before handing the object to `runRoutine`; the `roll assist stability` gate independently catches any tuning that makes the car unstable. |
| T-02-19 | Repudiation | a routine that reports `pass` unconditionally | mitigate | Every test case asserts the numeric value against its band, not just the boolean; and every gate has an explicit companion case proving it can fail (`ramp without assist`, `frictionSlip 10.5` skidpad, `bodyRollGain 0.20`). |
| T-02-02 | Tampering | DOM injection | accept | Nothing in `src/physics/telemetry/` touches the DOM; the browser presentation layer is plan 02-09's `src/debug/telemetry-hud.ts`. |
| T-02-SC | Tampering | npm/pip/cargo installs | mitigate | No install task; `@dimforge/rapier3d@0.20.0` pin asserted by the existing `tests/rapier-smoke.test.ts`. A pin bump invalidates every measured number in 02-RESEARCH.md and must trigger a full telemetry re-run. |
</threat_model>
